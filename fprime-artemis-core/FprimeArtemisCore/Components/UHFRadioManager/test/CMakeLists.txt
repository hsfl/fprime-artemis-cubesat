cmake_minimum_required(VERSION 3.16)
project(ArtemisRadioLinkTests LANGUAGES CXX)
enable_testing()
add_executable(radio_link_test LinkCoreTest.cpp)
target_compile_features(radio_link_test PRIVATE cxx_std_14)
if (MSVC)
    target_compile_options(radio_link_test PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(radio_link_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME radio_link_contract COMMAND radio_link_test)
add_executable(radio_channel_test ChannelLinkTest.cpp)
target_compile_features(radio_channel_test PRIVATE cxx_std_14)
if (MSVC)
    target_compile_options(radio_channel_test PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(radio_channel_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME radio_channel_contract COMMAND radio_channel_test)
